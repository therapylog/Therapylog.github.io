// api/launch-notify.js // ============================================================ // TherapyLog launch-notify capture → api.therapylog.app/api/launch-notify // ============================================================ // Accepts an email from the "notify me at launch" / newsletter signup // and adds it as a contact in a Resend audience. Used for the launch // waitlist and the monthly newsletter list. // // Vercel env vars: // RESEND_API_KEY (required — from resend.com/api-keys) // RESEND_AUDIENCE_ID (required — the audience to add contacts to) // ------------------------------------------------------------ function setCors(res, origin) { res.setHeader("Access-Control-Allow-Origin", origin); res.setHeader("Access-Control-Allow-Methods", "POST, OPTIONS"); res.setHeader("Access-Control-Allow-Headers", "Content-Type"); } function originAllowed(origin) { if (!origin) return true; if (origin.startsWith("https://therapylog.app")) return true; if (origin.startsWith("https://www.therapylog.app")) return true; if (origin === "capacitor://localhost") return true; if (origin === "http://localhost") return true; if (origin === "https://localhost") return true; return false; } // Basic, permissive email shape check. Not a validator — just a sanity gate. function looksLikeEmail(s) { return typeof s === "string" && /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(s.trim()) && s.length <= 254; } module.exports = async (req, res) => { const origin = req.headers.origin || ""; const allowed = originAllowed(origin); setCors(res, allowed ? (origin || "https://therapylog.app") : "https://therapylog.app"); if (req.method === "OPTIONS") return res.status(200).end(); if (req.method !== "POST") return res.status(405).json({ error: "Method not allowed" }); if (!allowed) return res.status(403).json({ error: "Origin not allowed" }); let data; try { data = typeof req.body === "string" ? JSON.parse(req.body) : req.body; } catch (e) { return res.status(400).json({ error: "Invalid request body" }); } const email = data && data.email ? String(data.email).trim().toLowerCase() : ""; const source = data && data.source ? String(data.source).slice(0, 40) : "web"; if (!looksLikeEmail(email)) { return res.status(400).json({ error: "Please enter a valid email address." }); } if (!process.env.RESEND_API_KEY || !process.env.RESEND_AUDIENCE_ID) { console.error("launch-notify: Resend env vars not configured"); return res.status(500).json({ error: "Signups aren't configured yet. Email hello@therapylog.app" }); } try { const r = await fetch( `https://api.resend.com/audiences/${process.env.RESEND_AUDIENCE_ID}/contacts`, { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${process.env.RESEND_API_KEY}` }, body: JSON.stringify({ email: email, unsubscribed: false }) } ); const out = await r.json().catch(() => ({})); // Resend returns an error if the contact already exists — treat that as success. if (!r.ok) { const msg = (out && out.message) ? String(out.message).toLowerCase() : ""; if (msg.includes("already") || r.status === 409) { return res.status(200).json({ ok: true, already: true }); } console.error("Resend contact error:", r.status, out); return res.status(502).json({ error: "Couldn't save your email right now. Please try again." }); } return res.status(200).json({ ok: true, source: source }); } catch (e) { console.error("launch-notify fetch failed:", e); return res.status(502).json({ error: "Couldn't reach the signup service. Please try again." }); } };
