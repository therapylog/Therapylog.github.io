// api/create-pro-subscription.js // ============================================================ // TherapyLog Stripe checkout → api.therapylog.app/api/create-pro-subscription // ============================================================ // Creates a Stripe Checkout session for any of the four subscription // plans, attaches affiliate ref metadata, and auto-applies the annual // coupon when the customer picks an annual plan. // // Vercel env vars: // STRIPE_SECRET_KEY (required) // STRIPE_PRO_MONTHLY_PRICE price_1Tfw16... ($9.99/mo) // STRIPE_PRO_ANNUAL_PRICE price_1Tfw16... ($99.99/yr) // STRIPE_BYOK_MONTHLY_PRICE price_1Tfw2d... ($8.99/mo) // STRIPE_BYOK_ANNUAL_PRICE price_1Tfw35... ($89.99/yr) // STRIPE_ANNUAL_COUPON_ID (optional — coupon ID for 1 month free // on annual plans; if not set, no discount // is auto-applied) // // Request body: // { plan: "pro_monthly" | "pro_annual" | "byok_monthly" | "byok_annual", // ref: "affiliatecode" (optional, stored in subscription metadata) } // // Response: // { url: "https://checkout.stripe.com/..." } — redirect the user here // ------------------------------------------------------------ const STRIPE_API = "https://api.stripe.com/v1"; // Map incoming plan keys to env var names const PRICE_MAP = { pro_monthly: "STRIPE_PRO_MONTHLY_PRICE", pro_annual: "STRIPE_PRO_ANNUAL_PRICE", byok_monthly: "STRIPE_BYOK_MONTHLY_PRICE", byok_annual: "STRIPE_BYOK_ANNUAL_PRICE", // Legacy keys from the old two-plan endpoint — keep working pro: "STRIPE_PRO_MONTHLY_PRICE", annual: "STRIPE_PRO_ANNUAL_PRICE", }; const ANNUAL_PLANS = new Set(["pro_annual", "byok_annual", "annual"]); // Tier label stored in metadata — tells payout scripts which tier was purchased const TIER_LABEL = { pro_monthly: "pro", pro_annual: "pro", byok_monthly: "byok", byok_annual: "byok", pro: "pro", annual: "pro", }; function setCors(res, origin) { res.setHeader("Access-Control-Allow-Origin", origin); res.setHeader("Access-Control-Allow-Methods", "POST, OPTIONS"); res.setHeader("Access-Control-Allow-Headers", "Content-Type"); } function originAllowed(origin) { if (!origin) return true; if (origin.startsWith("https://therapylog.app")) return true; if (origin.startsWith("https://www.therapylog.app")) return true; if (origin === "capacitor://localhost") return true; if (origin === "http://localhost") return true; if (origin === "https://localhost") return true; return false; } // Stripe API helper — urlencoded POST (Stripe's native format) async function stripePost(path, params, secretKey) { const body = Object.entries(params) .flatMap(([k, v]) => { if (v === null || v === undefined) return []; if (typeof v === "object") { // One level of nesting (e.g. metadata[ref]=..., line_items[0][price]=...) return Object.entries(v).map(([k2, v2]) => { if (v2 === null || v2 === undefined) return null; return `${encodeURIComponent(k)}[${encodeURIComponent(k2)}]=${encodeURIComponent(v2)}`; }).filter(Boolean); } return [`${encodeURIComponent(k)}=${encodeURIComponent(v)}`]; }) .join("&"); const r = await fetch(`${STRIPE_API}${path}`, { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded", "Authorization": `Bearer ${secretKey}`, "Stripe-Version": "2024-06-20" }, body }); return { ok: r.ok, status: r.status, data: await r.json() }; } module.exports = async (req, res) => { const origin = req.headers.origin || ""; const allowed = originAllowed(origin); setCors(res, allowed ? (origin || "https://therapylog.app") : "https://therapylog.app"); if (req.method === "OPTIONS") return res.status(200).end(); if (req.method !== "POST") return res.status(405).json({ error: "Method not allowed" }); if (!allowed) return res.status(403).json({ error: "Origin not allowed" }); if (!process.env.STRIPE_SECRET_KEY) { console.error("create-pro-subscription: STRIPE_SECRET_KEY not set"); return res.status(500).json({ error: "Checkout is not configured. Email hello@therapylog.app" }); } let data; try { data = typeof req.body === "string" ? JSON.parse(req.body) : req.body; } catch (e) { return res.status(400).json({ error: "Invalid request body" }); } const plan = data && data.plan ? String(data.plan).trim().toLowerCase() : null; const ref = data && data.ref ? String(data.ref).trim().toLowerCase().replace(/[^a-z0-9_-]/g, "").slice(0, 64) : ""; // Resolve price ID const priceEnvKey = plan ? PRICE_MAP[plan] : null; if (!priceEnvKey) { return res.status(400).json({ error: `Unknown plan: ${plan}. Valid plans: pro_monthly, pro_annual, byok_monthly, byok_annual` }); } const priceId = process.env[priceEnvKey]; if (!priceId) { console.error(`create-pro-subscription: env var ${priceEnvKey} not set`); return res.status(500).json({ error: "This plan isn't available yet. Email hello@therapylog.app" }); } const isAnnual = ANNUAL_PLANS.has(plan); const tier = TIER_LABEL[plan] || "pro"; // Success URL auto-activates the correct tier in the app so subscribers // don't have to manually click "Activate" after checkout. const successUrl = `https://therapylog.app/app?tl_activated=${tier}&session_id={CHECKOUT_SESSION_ID}`; const cancelUrl = `https://therapylog.app/pro`; // Build the Stripe Checkout session params. // Stripe's API uses form-encoded nested params, so we construct them carefully. const sessionParams = { mode: "subscription", "line_items[0][price]": priceId, "line_items[0][quantity]": "1", success_url: successUrl, cancel_url: cancelUrl, // Store ref and tier in subscription metadata for payout tracking "subscription_data[metadata][ref]": ref || "", "subscription_data[metadata][tier]": tier, "subscription_data[metadata][plan]": plan, }; if (isAnnual) { if (ref && process.env.STRIPE_ANNUAL_COUPON_ID) { // Affiliate-referred annual — auto-apply the 1-month-free coupon. // (discounts[] and allow_promotion_codes are mutually exclusive in Stripe Checkout) sessionParams["discounts[0][coupon]"] = process.env.STRIPE_ANNUAL_COUPON_ID; } else { // Direct annual, no affiliate link — customer can type a promo code // manually at checkout if they have one (keeps Phase A affiliate codes working). sessionParams.allow_promotion_codes = "true"; } } else { // Monthly — allow promo code entry for Phase A affiliate code tracking. sessionParams.allow_promotion_codes = "true"; } try { const result = await stripePost("/checkout/sessions", sessionParams, process.env.STRIPE_SECRET_KEY); if (!result.ok) { console.error("Stripe session error:", result.status, result.data); const msg = result.data && result.data.error && result.data.error.message ? result.data.error.message : "Could not create checkout session."; return res.status(result.status || 502).json({ error: msg }); } const url = result.data && result.data.url; if (!url) { console.error("Stripe returned no URL:", result.data); return res.status(502).json({ error: "Checkout setup failed. Please try again." }); } return res.status(200).json({ url }); } catch (e) { console.error("Stripe fetch failed:", e); return res.status(502).json({ error: "Could not reach the payment service. Please try again." }); } };
